cmake_minimum_required(VERSION 3.22)
project(kprobe CXX)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(KinematicProbe SHARED KinematicProbe.cc)
target_link_libraries(KinematicProbe gz-sim8::gz-sim8 gz-plugin2::register)
set_target_properties(KinematicProbe PROPERTIES CXX_STANDARD 17)
add_library(KinematicRing SHARED KinematicRing.cc)
target_link_libraries(KinematicRing gz-sim8::gz-sim8 gz-plugin2::register)
set_target_properties(KinematicRing PROPERTIES CXX_STANDARD 17)
